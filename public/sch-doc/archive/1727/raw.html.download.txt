  function dealTag(size, zoomObj, tagName) {
      var tagObj = zoomObj.getElementsByTagName(tagName);
      if (tagObj.length == 0) {
          zoomObj.style = "font-size:" + size + 'px !important';
      } else {
          for (var i = 0; i < tagObj.length; i++) {
              tagObj[i].style = "font-size:" + size + 'px !important';
          }
      }
  }
