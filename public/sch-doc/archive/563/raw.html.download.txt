<script>
  (function() {
    const win = document.getElementById('autoFloatingWindow');
    if (!win) return;

    let x = 100;
    let y = 100;
    let dx = 0.3;
    let dy = 1.0;
    let animationFrameId;
    let isVisible = true;

    // 关闭功能
    const closeBtn = document.getElementById('closeBtn');
    if(closeBtn) {
        closeBtn.addEventListener('click', () => {
            win.style.display = 'none';
            if(animationFrameId) cancelAnimationFrame(animationFrameId);
        });
    }

    // 监听页面可见性
    document.addEventListener("visibilitychange", () => {
        if (document.hidden) {
            isVisible = false;
            if(animationFrameId) cancelAnimationFrame(animationFrameId);
        } else {
            isVisible = true;
            animate();
        }
    });

    function animate() {
      if (!isVisible || win.style.display === 'none') return;

      const windowWidth = window.innerWidth;
      const windowHeight = window.innerHeight;
      const winRect = win.getBoundingClientRect(); // 改用 getBoundingClientRect 获取真实渲染尺寸
      const winWidth = winRect.width;
      const winHeight = winRect.height;

      x += dx;
      y += dy;

      // 边界检测（增加了安全余量，防止极端情况下溢出）
      if (x <= 0 || x + winWidth >= windowWidth) {
        dx = -dx;
        x = Math.max(0, Math.min(x, windowWidth - winWidth));
      }
      if (y <= 0 || y + winHeight >= windowHeight) {
        dy = -dy;
        y = Math.max(0, Math.min(y, windowHeight - winHeight));
      }

      win.style.transform = `translate3d(${x}px, ${y}px, 0)`;
      animationFrameId = requestAnimationFrame(animate);
    }

    // 启动
    animate();
  })();
</script>
