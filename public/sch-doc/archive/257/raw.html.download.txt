    $(function(){
        show();

            //     $('#a_left').addClass('disabled');
            // }else{
            //     $('#a_left').removeClass('disabled');
            // }
            show(id);

        $('#a_right').click(function(){
            id++;
            $('#a_left').removeClass('disabled');
            show(id);
        });

        $('#rl_info').on('click','.act,.act_h_k,.act_r',function(){
            var date = $(this).attr('data-date');
            if(date){
                $.ajax({
                    url:'https://scce.sjtu.edu.cn/active/show_days_articles.html',
                    data:{date:date},
                    type:'post',
                    dataType:'json',
                    success:function(data){
                        $('#xsrl_html').html(data.res);
                    }
                });
            }
            
        })

    })
    function show(i=0){
        $.ajax({
            url:'https://scce.sjtu.edu.cn/active/show_days.html',
            data:{'num':i},
            type:'post',
            dataType:'json',
            success:function(data){
                $('.day').html(data.day);
                $('#rl_info').html(data.res);
            }
        });
    }
</script>
