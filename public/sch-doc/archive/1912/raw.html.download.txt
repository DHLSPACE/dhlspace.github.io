                <div class="trs_editor_view TRS_UEDITOR trs_paper_default trs_word"><p style="font-size: 16px; line-height: 2em;"><span data-index="9" style="line-height: 150%; font-family: arial, helvetica, sans-serif; font-size: 16px;">各位考生：</span></p><p style="text-indent: 28px; font-size: 16px; line-height: 2em;"><span data-index="9" style="line-height: 150%; font-family: arial, helvetica, sans-serif; font-size: 16px;">请按复试日程表的时间要求按时报到，一志愿报考考生无论是否接到复试的电话通知，只要单科和总分在公布的复试线以上，就务必按网上公告的要求来中国科学技术大学研究生院科学岛分院参加复试。</span></p><p style="text-indent: 28px; font-size: 16px; line-height: 2em;"><span data-index="9" style="line-height: 150%; font-family: arial, helvetica, sans-serif; font-size: 16px;">请达线考生于2025年3月19日12时前，加QQ群：1023632431（群号），以便及时了解相关信息。</span></p><p style="text-indent: 28px; font-size: 16px; line-height: 2em;"><strong><span style="font-family: arial, helvetica, sans-serif; text-indent: 0em;">注：入群密码通过短信告知考生。</span></strong></p><p style="font-size: 16px; line-height: 2em;"><span data-index="9" style="font-family: arial, helvetica, sans-serif; font-size: 16px;"><strong><span style="line-height: 150%; font-family: arial, helvetica, sans-serif; font-size: 16px;">一、报考材料提交</span></strong></span></p><p style="text-indent: 28px; font-size: 16px; line-height: 2em;"><span data-index="9" style="line-height: 150%; font-family: arial, helvetica, sans-serif; font-size: 16px;">按照教育部的文件精神在复试时须查验考生的报考资格,请各位考生在报到时务必提交以下材料并按所列顺序用钉书针装订成册,以便于对你们的报考资格进行审查。</span></p><p style="text-indent: 28px; font-size: 16px; line-height: 2em;"><span data-index="9" style="line-height: 150%; font-family: arial, helvetica, sans-serif; font-size: 16px;">1、请于2025年3月19日18时以后，用考生本人报考的手机号码和默认密码（身份证号码）登录科学岛分院研究生信息系统：https://gs.hfcas.ac.cn/（请及时修改自己的密码），填报《中国科学技术大学研究生院科学岛分院2025年招收攻读硕士学位研究生考生个人简况及自述》，自行下载并打印和签字；</span></p><p style="text-indent: 28px; font-size: 16px; line-height: 2em;"><span data-index="9" style="line-height: 150%; font-family: arial, helvetica, sans-serif; font-size: 16px;">2、本人签名的《诚信复试承诺书》（附件1）原件；</span></p><p style="text-indent: 28px;                 const compact = text.match(/(\d{4})(\d{2})(\d{2})/);
                if (compact) return [Number(compact[1]), Number(compact[2]), Number(compact[3])];
                const dotted = text.match(/(\d{4})\.(\d{1,2})\.(\d{1,2})/);
                if (dotted) return [Number(dotted[1]), Number(dotted[2]), Number(dotted[3])];
                const dated = text.match(/(\d{4})年(\d{1,2})月(\d{1,2})日/);
                if (dated) return [Number(dated[1]), Number(dated[2]), Number(dated[3])];
                const dash1 = text.match(/(\d{4})-(\d{1,2})-(\d{1,2})/);
                if (dash1) return [Number(dash1[1]), Number(dash1[2]), Number(dash1[3])];
                const dash2 = text.match(/(\d{4})-(\d{2})-(\d{2})/);
                if (dash2) return [Number(dash2[1]), Number(dash2[2]), Number(dash2[3])];

            const titleEl = document.querySelector(".article-title-box");
            const timeEl = document.querySelector(".article-detail-time");
            
            if (titleEl && timeEl) {
                const timeAttr = timeEl.textContent || "";
                const hhmmss = (timeAttr.match(/(\d{2}:\d{2}:\d{2})/) || ["23:59:59"])[0];

                console.log("Checking 公示时间 for:", originalText, "with timeAttr:", timeAttr);

                if (!originalText.startsWith("[公示已结束]")) {

                    if (parts) {

                        const [h, min, s] = hhmmss.split(":").map(Number);
                        const combined = new Date(y, m - 1, d, h, min, s);
                        if (combined.getTime() < Date.now()) {

                        }
                    }
                }
            }
        }
