$(document).ready(function() {
    var index = 0; // 初始化索引
    var liElements = $(".container-sec .index_tab .tabmenu li"); // 获取所有的li元素
    var tabConElements = $(".container-sec .tabox .tab_con"); // 获取所有的tab_con元素
    var totalLis = liElements.length; // 获取li元素的总数
    var intervalId; // 用于存储定时器的引用
    var isAnyMouseOver = false; // 用于跟踪鼠标是否处于li元素或tab_con元素上
 
    // 启动定时器
    function startInterval() {
        if (!isAnyMouseOver) {
            intervalId = setInterval(function() {
                // 更新UI
                updateUI();
            }, 3000); // 3000毫秒即3秒
        }
    }
 
    // 停止定时器
    function stopInterval() {
        clearInterval(intervalId);
    }
 
    // 更新UI：根据当前索引设置selected类和显示对应的tab_con
    function updateUI() {
        liElements.removeClass("selected");
        tabConElements.hide();
        liElements.eq(index).addClass("selected");
        tabConElements.eq(index).show();
        index = (index + 1) % totalLis;
    }
 
    // 为li元素添加鼠标事件
    liElements.on('mouseenter mouseleave', function(event) {
        isAnyMouseOver = event.type === 'mouseenter';
        handleMouseOverStateChange();
    });
 
    // 为tab_con元素添加鼠标事件
    tabConElements.on('mouseenter mouseleave', function(event) {
        isAnyMouseOver = event.type === 'mouseenter';
        handleMouseOverStateChange();
    });
 
    // 处理鼠标移入/移出状态变化
    function handleMouseOverStateChange() {
        if (isAnyMouseOver) {
            stopInterval(); // 鼠标移入时停止定时器
            // 可选：立即更新UI以显示当前鼠标所在的元素（如果需要的话）
            // 但由于鼠标可能只是在元素边缘快速移动，这通常不是必需的
            // 可以在需要时调用 updateUIWithCurrentMousePosition() 函数
        } else {
            startInterval(); // 鼠标移出时启动定时器
        }
    }
 
    // 初始化时启动定时器
    startInterval();
 
    // 可选的辅助函数（如果需要的话）：

    // }
});

	String.prototype.replaceAll = function(s1, s2) {
		return this.replace(new RegExp(s1, "gm"), s2);
	}
	var reg = /<\/?.+?\/?>/g;
	var abstract = "";
	var content = "";
	var formatContent = "";
	/**
	 * 判断摘要为空取正文
	 * @param  {[object]} {
	 * .item	 每条新闻元素的class
	 * .abstract 摘要盒子
	 * .content  正文盒子
	 * }
	 * @return {[type]}
	 */
	$(".itemtxt").each(function(){
		abstract = $(this).find(".abstract").text().trim();
		content = $(this).find(".content").text().trim();
		formatContent = content.replaceAll(' ',' ');
		formatContent = formatContent.replace(reg,'');
                formatContent = formatContent.replaceAll('&nbsp;','');
                formatContent = formatContent.replaceAll('　','');
                formatContent = formatContent.replaceAll('？','');
		if(abstract==""){
			$(this).find(".abstract").remove();
		}else{
			$(this).find(".content").remove();
		}
		if(content==""){
			$(this).find(".content").remove();
		}else{
			$(this).find(".content").html(formatContent);
		}
	});
