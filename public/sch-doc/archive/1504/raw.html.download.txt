	    	 /**
	         * 合并单元格
	         * @param tableId   表格的ID
	         * @param startRow  起始行
	         * @param endRow    结束行:0代表合并所有行
	         * @param colIndex  合并的列号，对第几列进行合并(从0开始)。第一行从0开始
	         * @param mergeBlank是否合并空值单元格，默认不合并
	         */
			function mergeCells(tableId, startRow, endRow, colIndex, mergeBlank) {
	    		var tb = document.getElementById(tableId);
	            if(!tb || !tb.rows || tb.rows.length <= 0) {
	                return;
	            }
	            if(colIndex >= tb.rows[0].cells.length || (startRow >= endRow && endRow != 0)) {
	                return;
	            }
	            if(endRow == 0) {
	                endRow = tb.rows.length - 1;
	            }
	            for(var i = startRow; i < endRow; i++) {
	            	if((!mergeBlank && (!tb.rows[startRow].cells[colIndex] || tb.rows[startRow].cells[colIndex].innerHTML == '')) 
	            			|| (!tb.rows[startRow].cells[colIndex] || !tb.rows[i + 1].cells[colIndex])) {
	            		mergeCells(tableId, i + 1, endRow, colIndex, mergeBlank);
	                    break;
	            	}

	                } else {
	                    mergeCells(tableId, i + 1, endRow, colIndex, mergeBlank);

	                } 
	            }

			function selectChange(ob,data) {
			        var obj = $(ob);
			        var content = $(ob).attr('data-placeholder');
			        var showfirst = $(ob).attr('showfirst');
			        var showdm= $(ob).attr('showdm');
			        var showEx= $(ob).attr('showEx');
			        var showid = $(ob).attr('id');
			        if ($(ob).attr("selecturl") == null) {
			        } else if($(ob).attr('multiple')==undefined) {
			            var type = 'get';
			            if ($(ob).attr("notcache") == 'true') {//不让缓存,每次请求新的数据
			                type = 'post';
			            }
			            var url=$(ob).attr("selecturl");
			            if ($(ob).attr("linkId")) {
			                linkSelect($(this).attr("linkId"),$(this).attr("id"))
			                url=url+"?"+ $("#" + $(this).attr("linkId")).attr("name") +"="+$("#" + $(this).attr("linkId")).val();
			            }
			            $.ajax({
			                type:type,
			                dataType:"json", // 返回json格式的数据
			                async:true,
			                url:url, // 要访问的后台地址
			                success:function (msg) {// msg为返回的数据，在这里做数据绑定
			                	console.log(msg);
			                    var options = '';
			                    if(showfirst!='true'){
				                    if (content != null && content != '') {
				                        options += '<option value="">' + content + '</option>';
				                    } else {
				                        	options += '<option value="">----请选择----</option>';
				                    }
			                    }
			                    
			                    var b = obj.attr("name");
			                    for (var i = 0; i < msg.length; i++) {
			                        if (msg[i].dataMap) { // 添加扩展字段dataMap
			                            var exFields = '';
			                            for (var key in msg[i].dataMap) {
			                                exFields += key + '="' + (msg[i].dataMap[key] ? msg[i].dataMap[key] : '') + '" ';
			                            }
			                        }
			                        var msgtext='';
			                        if(showdm=='true'){
			                        	msgtext= msg[i].value+' '+msg[i].text;
			                        }else{
			                        	msgtext= msg[i].text;
			                        }
			                        if(showEx=='true'){
			                        	if(msg[i].extraField1!=null&&msg[i].extraField1!=''){
			                        		msgtext= msg[i].value+' '+msg[i].text+"<font color='red'>(缺额人数:"+ msg[i].extraField1+")</font>";
			                        	}else{
			                        		msgtext= msg[i].value+' '+msg[i].text;
			                        	}
			                        }
			                        if ((data != null && data[obj.attr("name")] == msg[i].value)||(data != null && data[obj.attr("name")] == msg[i].text)) {
			                            options += '<option value="' + msg[i].value  + '" extra="'+msg[i].extraField1+'" selected="selected" ' + exFields + ' mc="'+msg[i].text+'">' + msgtext + "</option>";
			                        } else {
			                            options += '<option value="' + msg[i].value + '" extra="'+msg[i].extraField1+'" ' + exFields + ' mc="'+msg[i].text+'">' + msgtext + "</option>";
			                        }
			                    }
			                   
			                    
			                    obj.html(options);
			                },
			                error:function (XHR, status, errorThrow) {
			                    return false;
			                }
			            });

			        }
			}
