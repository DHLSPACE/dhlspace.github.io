// 移动端菜单功能 - 使用图标切换
document.addEventListener('DOMContentLoaded', function() {
    console.log('DOM loaded - initializing mobile menu with icon toggle');
    
    const menuButton = document.querySelector('button.lg\\:hidden');
    console.log('Menu button found:', menuButton);
    
    // 使用更准确的选择器找到移动菜单
    const mobileMenu = document.querySelector('.lg\\:hidden.bg-white');
    console.log('Mobile menu found:', mobileMenu);
    
    if (!menuButton || !mobileMenu) {
        console.error('Menu elements not found');
        return;
    }
    
    // 保存原始菜单图标HTML
    const menuIconHTML = `
        <svg xmlns="http://www.w3.org/2000/svg" width="28" height="28" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" class="lucide lucide-menu" aria-hidden="true">
            <path d="M4 5h16"></path>
            <path d="M4 12h16"></path>
            <path d="M4 19h16"></path>
        </svg>
    `;
    
    const closeIconHTML = `
        <svg xmlns="http://www.w3.org/2000/svg" width="28" height="28" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" class="lucide lucide-x" aria-hidden="true">
            <path d="M18 6 6 18"></path>
            <path d="m6 6 12 12"></path>
        </svg>
    `;
    
    // 初始隐藏移动菜单
    mobileMenu.style.display = 'none';
    
    // 切换菜单显示/隐藏和图标
    function toggleMenu() {
        console.log('Menu button clicked');
        
        if (mobileMenu.style.display === 'none') {
            // 打开菜单
            mobileMenu.style.display = 'block';
            document.body.style.overflow = 'hidden';
            // 切换为关闭图标
            menuButton.innerHTML = closeIconHTML;
        } else {
            // 关闭菜单
            mobileMenu.style.display = 'none';
            document.body.style.overflow = '';
            // 切换为菜单图标
            menuButton.innerHTML = menuIconHTML;
        }
    }
    
    // 关闭菜单函数
    function closeMenu() {
        console.log('Closing menu');
        mobileMenu.style.display = 'none';
        document.body.style.overflow = '';
        // 切换回菜单图标
        menuButton.innerHTML = menuIconHTML;
    }
    
    // 打开菜单函数
    function openMenu() {
        console.log('Opening menu');
        mobileMenu.style.display = 'block';
        document.body.style.overflow = 'hidden';
        // 切换为关闭图标
        menuButton.innerHTML = closeIconHTML;
    }
    
    // 绑定菜单按钮点击事件
    menuButton.addEventListener('click', function(e) {
        e.stopPropagation();
        toggleMenu();
    });
    
    // 点击菜单链接时关闭菜单
    const menuLinks = mobileMenu.querySelectorAll('a');
    menuLinks.forEach(link => {
        link.addEventListener('click', function() {
            closeMenu();
        });
    });
    
    // 点击菜单外部区域关闭菜单
    document.addEventListener('click', function(e) {
        if (mobileMenu.style.display === 'block' && 
            !mobileMenu.contains(e.target) && 
            e.target !== menuButton) {
            closeMenu();
        }
    });
    
    // 防止菜单内部点击事件冒泡
    mobileMenu.addEventListener('click', function(e) {
        e.stopPropagation();
    });
    
    // 响应式处理：窗口大小变化时，如果切换到桌面端，确保菜单关闭且图标重置
    window.addEventListener('resize', function() {
        if (window.innerWidth >= 1024) { // lg breakpoint
            closeMenu();
        }
    });
    
    console.log('Mobile menu with icon toggle initialized successfully');
});
