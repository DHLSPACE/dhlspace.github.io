<script>
	let pageNum = 1;
	let totalPage = 1;
	$(() => {
		getList()
		$("#pageInput").keypress(function(event) {
			var keycode = event.which || event.keyCode
			if (keycode == '13') {
				const val = $("#pageInput").val()
				if (val > 0 && val <= totalPage) {
					pageNum = val
					getList()
				}
			}
		});
		$("#prevBtn").click(function(event) {
			if (pageNum <= 1) {
				return
			} else {
				pageNum--
				getList()
			}
		});
		$("#nextBtn").click(function(event) {
			if (pageNum >= totalPage) {
				return
			} else {
				pageNum++
				getList()
			}
		});
	})
	const getList = function () {
		renderPaginationFn()
	}
	const renderPaginationFn = function () {
		$.each($('.link-item'), function(index, val) {
			$(val).removeClass('active')
			if (index + 1 == pageNum) {
				$(val).addClass('active')
			}
		})
		$.each($('.link-item'), function(index, val) {
			$(val).click(function (ele) {
				pageNum = $(ele.target).text()
				getList()
			})
		})
	}
</script>
